\subsection{Actividades por desarrollar}

\textbf{1) Reglas de negocio}
Las reglas de negocio deben redactarse antes de construir las tablas para dejar claras las restricciones del caso y la relación entre sus entidades.
\begin{itemize}
    \item Cada empresa puede tener varios ejecutivos, pero cada ejecutivo pertenece a una sola empresa.
    \item Cada hotel puede registrar varios recepcionistas y varias habitaciones.
    \item Cada habitación pertenece a un solo hotel y debe conservar un tipo, numero y precio definidos.
    \item Un hospedaje relaciona a un ejecutivo, una habitación y un recepcionista en un rango de fechas determinado.
    \item Un recepcionista puede registrar varios hospedajes, pero cada hospedaje debe ser atendido por un solo recepcionista.
    \item Una novedad debe estar asociada a un hospedaje existente.
    \item Los datos de prueba deben permitir validar consultas sobre uso de habitaciones, costos, empresas y novedades.
\end{itemize}

\textbf{2) Modelo Entidad-Relación}
El modelo entidad-relación integra las entidades Empresa, Ejecutivo, Hotel, Recepcionista, Habitación, Hospedaje y Novedad. Debe incluir la relación de hospedaje como el vínculo principal entre la operación hotelera y los ejecutivos, así como la entidad de novedades asociadas a cada hospedaje.

\textbf{2) Creación de tablas}
\begin{lstlisting}
CREATE TABLE empresa (
    id_empresa NUMBER PRIMARY KEY,
    nombre VARCHAR2(40) NOT NULL,
    direccion VARCHAR2(50) NOT NULL,
    telefono VARCHAR2(20) NOT NULL
);

CREATE TABLE ejecutivo (
    id_ejecutivo NUMBER PRIMARY KEY,
    nombre VARCHAR2(40) NOT NULL,
    apellido VARCHAR2(40) NOT NULL,
    cargo VARCHAR2(40) NOT NULL,
    id_empresa NUMBER,
    CONSTRAINT fk_ejecutivo_empresa
        FOREIGN KEY (id_empresa) REFERENCES empresa(id_empresa)
);

CREATE TABLE hotel (
    id_hotel NUMBER PRIMARY KEY,
    nombre VARCHAR2(40) NOT NULL,
    ciudad VARCHAR2(40) NOT NULL,
    direccion VARCHAR2(50) NOT NULL
);

CREATE TABLE recepcionista (
    id_recepcionista NUMBER PRIMARY KEY,
    nombre VARCHAR2(40) NOT NULL,
    turno VARCHAR2(20) NOT NULL,
    id_hotel NUMBER,
    CONSTRAINT fk_recepcionista_hotel
        FOREIGN KEY (id_hotel) REFERENCES hotel(id_hotel)
);

CREATE TABLE habitacion (
    id_habitacion NUMBER PRIMARY KEY,
    numero VARCHAR2(10) NOT NULL,
    tipo VARCHAR2(20) NOT NULL,
    precio NUMBER(10,2) NOT NULL,
    id_hotel NUMBER,
    CONSTRAINT fk_habitacion_hotel
        FOREIGN KEY (id_hotel) REFERENCES hotel(id_hotel)
);

CREATE TABLE hospedaje (
    id_hospedaje NUMBER PRIMARY KEY,
    id_ejecutivo NUMBER,
    id_habitacion NUMBER,
    id_recepcionista NUMBER,
    fecha_ingreso DATE NOT NULL,
    fecha_salida DATE NOT NULL,
    CONSTRAINT fk_hospedaje_ejecutivo
        FOREIGN KEY (id_ejecutivo) REFERENCES ejecutivo(id_ejecutivo),
    CONSTRAINT fk_hospedaje_habitacion
        FOREIGN KEY (id_habitacion) REFERENCES habitacion(id_habitacion),
    CONSTRAINT fk_hospedaje_recepcionista
        FOREIGN KEY (id_recepcionista) REFERENCES recepcionista(id_recepcionista)
);

CREATE TABLE novedad (
    id_novedad NUMBER PRIMARY KEY,
    descripcion CLOB NOT NULL,
    fecha DATE NOT NULL,
    id_hospedaje NUMBER,
    CONSTRAINT fk_novedad_hospedaje
        FOREIGN KEY (id_hospedaje) REFERENCES hospedaje(id_hospedaje)
);

CREATE SEQUENCE seq_empresa START WITH 1;
CREATE SEQUENCE seq_ejecutivo START WITH 1;
CREATE SEQUENCE seq_hotel START WITH 1;
CREATE SEQUENCE seq_recepcionista START WITH 1;
CREATE SEQUENCE seq_habitacion START WITH 1;
CREATE SEQUENCE seq_hospedaje START WITH 1;
CREATE SEQUENCE seq_novedad START WITH 1;

CREATE OR REPLACE TRIGGER trg_empresa

\textbf{4) Consultas avanzadas del Grupo F (Análisis por tipo de habitación)}

\textbf{Consulta 1: Tipo de habitación más utilizado por CNT (frecuencia mayor al promedio)}
\begin{lstlisting}
SELECT h.tipo
FROM habitacion h
JOIN hospedaje ho ON h.id_habitacion = ho.id_habitacion
JOIN ejecutivo e ON ho.id_ejecutivo = e.id_ejecutivo
JOIN empresa em ON e.id_empresa = em.id_empresa
WHERE em.nombre = 'CNT'
GROUP BY h.tipo
HAVING COUNT(*) > (
    SELECT AVG(freq)
    FROM (
        SELECT COUNT(*) freq
        FROM habitacion h2
        JOIN hospedaje ho2 ON h2.id_habitacion = ho2.id_habitacion
        JOIN ejecutivo e2 ON ho2.id_ejecutivo = e2.id_ejecutivo
        JOIN empresa em2 ON e2.id_empresa = em2.id_empresa
        WHERE em2.nombre = 'CNT'
        GROUP BY h2.tipo
    ) sub
);
\end{lstlisting}

\textbf{Consulta 2: Hotel cuya habitación más cara fue usada por ejecutivos de más de una empresa}
\begin{lstlisting}
SELECT *
FROM hotel h
WHERE h.id_hotel IN (
    SELECT ha.id_hotel
    FROM habitacion ha
    WHERE ha.precio = (SELECT MAX(precio) FROM habitacion)
            AND ha.id_habitacion IN (
                SELECT ho.id_habitacion
                FROM hospedaje ho
                JOIN ejecutivo e ON ho.id_ejecutivo = e.id_ejecutivo
                GROUP BY ho.id_habitacion
                HAVING COUNT(DISTINCT e.id_empresa) > 1
            )
);
\end{lstlisting}

\textbf{Consulta 3: Ejecutivo hospedado en suite con costo total mayor que todas las simples de marzo 2026}
\begin{lstlisting}
SELECT e.nombre, e.apellido
FROM ejecutivo e
WHERE e.id_ejecutivo IN (
    SELECT ho.id_ejecutivo
    FROM hospedaje ho
    JOIN habitacion h ON ho.id_habitacion = h.id_habitacion
        WHERE h.tipo = 'Suite'
            AND (ho.fecha_salida - ho.fecha_ingreso) * h.precio > ALL (
                SELECT (ho2.fecha_salida - ho2.fecha_ingreso) * h2.precio
                FROM hospedaje ho2
                JOIN habitacion h2 ON ho2.id_habitacion = h2.id_habitacion
                WHERE h2.tipo = 'Individual'
                    AND EXTRACT(MONTH FROM ho2.fecha_ingreso) = 3
                    AND EXTRACT(YEAR FROM ho2.fecha_ingreso) = 2026
            )
);
\end{lstlisting}

\textbf{Consulta 4: Recepcionista con más hospedajes en habitaciones matrimoniales de su hotel}
\begin{lstlisting}
SELECT *
FROM recepcionista r
WHERE r.id_recepcionista IN (
    SELECT ho.id_recepcionista
    FROM hospedaje ho
    JOIN habitacion h ON ho.id_habitacion = h.id_habitacion
    WHERE h.tipo = 'Matrimonial'
    GROUP BY ho.id_recepcionista, h.id_hotel
    HAVING COUNT(*) >= ALL (
        SELECT COUNT(*)
        FROM hospedaje ho2
        JOIN habitacion h2 ON ho2.id_habitacion = h2.id_habitacion
        WHERE h2.tipo = 'Matrimonial'
          AND h2.id_hotel = h.id_hotel
        GROUP BY ho2.id_recepcionista
    )
);
\end{lstlisting}

\textbf{Consulta 5: Empresa cuyos ejecutivos usaron todos los tipos de habitación}
\begin{lstlisting}
SELECT *
FROM empresa em
WHERE NOT EXISTS (
    SELECT tipo
    FROM habitacion
    MINUS
    SELECT DISTINCT h.tipo
    FROM hospedaje ho
    JOIN habitacion h ON ho.id_habitacion = h.id_habitacion
    JOIN ejecutivo e ON ho.id_ejecutivo = e.id_ejecutivo
    WHERE e.id_empresa = em.id_empresa
);
\end{lstlisting}

\textbf{Consulta 6: Habitación con menor hospedaje en su hotel y sin novedades}

SELECT *
FROM habitacion h
WHERE h.id_habitacion IN (
    SELECT ho.id_habitacion
    FROM hospedaje ho
    JOIN habitacion h2 ON ho.id_habitacion = h2.id_habitacion
    GROUP BY ho.id_habitacion, h2.id_hotel
    HAVING COUNT(*) <= ALL (
        SELECT COUNT(*)
        FROM hospedaje ho2
        WHERE ho2.id_habitacion IN (
            SELECT id_habitacion
            FROM habitacion
            WHERE id_hotel = h.id_hotel
        )
        GROUP BY ho2.id_habitacion
    )
)
AND h.id_habitacion NOT IN (
    SELECT ho.id_habitacion
    FROM hospedaje ho
    JOIN novedad n ON ho.id_hospedaje = n.id_hospedaje
);
\end{lstlisting}

Resultados principales:

\subsection{Defensa práctica}
\textbf{Consulta 5: conjunto de tipos existentes}

El conjunto completo se obtiene con:
\begin{lstlisting}
SELECT DISTINCT tipo FROM habitacion;
\end{lstlisting}

Ese conjunto se compara contra los tipos usados por cada empresa con NOT EXISTS para comprobar cobertura total.

\textbf{Consulta 3: subconsulta de comparación (prueba aislada)}

Subconsulta aislada para verificar costos de habitaciones individuales de marzo 2026:
\begin{lstlisting}
SELECT (ho2.fecha_salida - ho2.fecha_ingreso) * h2.precio AS costo_total_simple
FROM hospedaje ho2
JOIN habitacion h2 ON ho2.id_habitacion = h2.id_habitacion
WHERE h2.tipo = 'Individual'
    AND EXTRACT(MONTH FROM ho2.fecha_ingreso) = 3
    AND EXTRACT(YEAR FROM ho2.fecha_ingreso) = 2026;
\end{lstlisting}

\textbf{Consulta 4: versión con ALL y versión con MAX}

Versión con ALL:
\begin{lstlisting}
SELECT ho.id_recepcionista, h.id_hotel
FROM hospedaje ho
JOIN habitacion h ON ho.id_habitacion = h.id_habitacion
WHERE h.tipo = 'Matrimonial'
GROUP BY ho.id_recepcionista, h.id_hotel
HAVING COUNT(*) >= ALL (
        SELECT COUNT(*)
        FROM hospedaje ho2
        JOIN habitacion h2 ON ho2.id_habitacion = h2.id_habitacion
        WHERE h2.tipo = 'Matrimonial'
            AND h2.id_hotel = h.id_hotel
        GROUP BY ho2.id_recepcionista
);
\end{lstlisting}

Versión con subconsulta MAX:
\begin{lstlisting}
SELECT ho.id_recepcionista, h.id_hotel
FROM hospedaje ho
JOIN habitacion h ON ho.id_habitacion = h.id_habitacion
WHERE h.tipo = 'Matrimonial'
GROUP BY ho.id_recepcionista, h.id_hotel
HAVING COUNT(*) = (
        SELECT MAX(cnt)
        FROM (
                SELECT COUNT(*) AS cnt
                FROM hospedaje ho2
                JOIN habitacion h2 ON ho2.id_habitacion = h2.id_habitacion
                WHERE h2.tipo = 'Matrimonial'
                    AND h2.id_hotel = h.id_hotel
                GROUP BY ho2.id_recepcionista
        ) t
);
\end{lstlisting}

Comparación breve:
\begin{itemize}
        \item ALL es directo para comparaciones contra todo el conjunto.
        \item MAX es más legible cuando se desea mostrar explícitamente el valor máximo de referencia.
\end{itemize}

\textbf{Casos de prueba necesarios para que la consulta 6 retorne datos}
\begin{itemize}
        \item Debe existir al menos una habitación sin registros en novedad.
        \item Debe haber diferencia de frecuencias de hospedaje dentro de un mismo hotel (o empate mínimo válido).
        \item La habitación candidata debe pertenecer al hotel evaluado y tener la menor frecuencia relativa en ese hotel.
\end{itemize}
